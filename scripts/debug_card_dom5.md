# trpcState에서 프롬프트 추출

```js
(function(){var imgs=document.querySelectorAll('img');var g=[];for(var i=0;i<imgs.length;i++){if(imgs[i].src&&imgs[i].src.includes('getMediaUrlRedirect')){g.push(imgs[i])}}var a=g[0].closest('a');var href=a.href;var editId=href.split('/edit/')[1];console.log('editId:',editId);fetch(href,{credentials:'include'}).then(function(r){return r.text()}).then(function(html){var m=html.match(/<script id="__NEXT_DATA__"[^>]*>([\s\S]*?)<\/script>/);var data=JSON.parse(m[1]);var trpc=data.props.pageProps.trpcState;console.log('trpcState 키:',Object.keys(trpc));var queries=trpc.json?.queries||trpc.queries||[];console.log('queries 수:',queries.length);for(var q=0;q<queries.length;q++){var qk=JSON.stringify(queries[q].queryKey||'').substring(0,100);var hasData=queries[q].state?.data!=null;console.log('query['+q+'] key='+qk+' hasData='+hasData);if(hasData){var d=queries[q].state.data;var ds=JSON.stringify(d);if(ds.includes(editId)){console.log('  *** editId 발견! ***');console.log('  data 키:',typeof d==='object'?Object.keys(d):'primitive');if(ds.length<2000){console.log('  전체:',ds)}else{var idx=ds.indexOf(editId);console.log('  주변:',ds.substring(Math.max(0,idx-300),idx+500))}}}}}).catch(function(e){console.error('실패:',e)})})();
```
