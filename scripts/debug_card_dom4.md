# __NEXT_DATA__에서 프롬프트 추출

```js
(function(){var imgs=document.querySelectorAll('img');var g=[];for(var i=0;i<imgs.length;i++){if(imgs[i].src&&imgs[i].src.includes('getMediaUrlRedirect')){g.push(imgs[i])}}var a=g[0].closest('a');var href=a.href;console.log('fetch:',href);fetch(href,{credentials:'include'}).then(function(r){return r.text()}).then(function(html){var m=html.match(/<script id="__NEXT_DATA__"[^>]*>([\s\S]*?)<\/script>/);if(!m){console.log('__NEXT_DATA__ 못찾음');return}var data=JSON.parse(m[1]);console.log('__NEXT_DATA__ 최상위 키:',Object.keys(data));console.log('props 키:',Object.keys(data.props||{}));console.log('pageProps 키:',Object.keys(data.props?.pageProps||{}));var str=JSON.stringify(data);console.log('전체 길이:',str.length);var editId=href.split('/edit/')[1];if(str.includes(editId)){var idx=str.indexOf(editId);console.log('editId 주변:',str.substring(Math.max(0,idx-200),idx+300))}var prompts=[];var regex=/"(?:prompt|text|description|query|input_text)":\s*"([^"]{20,})"/g;var pm;while((pm=regex.exec(str))!==null){prompts.push(pm[0])}if(prompts.length>0){console.log('프롬프트 후보 '+prompts.length+'개:');for(var p=0;p<Math.min(10,prompts.length);p++){console.log('  ',prompts[p].substring(0,200))}}else{console.log('프롬프트 패턴 못찾음. 긴 문자열 검색...');var longStr=[];var lr=/"[a-zA-Z_]+":"([^"]{50,})"/g;var lm;while((lm=lr.exec(str))!==null){if(!lm[1].startsWith('http')&&!lm[1].startsWith('/')&&!lm[1].includes('<')){longStr.push(lm[0])}}console.log('긴 문자열 '+longStr.length+'개:');for(var l=0;l<Math.min(15,longStr.length);l++){console.log('  ',longStr[l].substring(0,200))}}}).catch(function(e){console.error('실패:',e)})})();
```
