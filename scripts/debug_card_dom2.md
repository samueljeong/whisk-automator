# Flow DOM 심층 진단 #2

이미지의 링크, 속성, URL 패턴 등을 분석합니다. Flow 탭 콘솔에 붙여넣기:

```js
(function(){var imgs=document.querySelectorAll('img');var g=[];for(var i=0;i<imgs.length;i++){if(imgs[i].src&&imgs[i].src.includes('getMediaUrlRedirect')){g.push(imgs[i])}}console.log('=== 이미지 '+g.length+'개 심층 분석 ===');for(var gi=0;gi<Math.min(5,g.length);gi++){var img=g[gi];console.log('\n--- 이미지 #'+(gi+1)+' ---');console.log('FULL src:',img.src);console.log('alt:',img.alt);console.log('title:',img.title);console.log('data attrs:',JSON.stringify(Object.keys(img.dataset)));console.log('aria:',img.getAttribute('aria-label'),img.getAttribute('aria-describedby'));var a=img.closest('a');if(a){console.log('링크 href:',a.href);console.log('링크 data:',JSON.stringify(Object.keys(a.dataset)));console.log('링크 aria:',a.getAttribute('aria-label'),a.getAttribute('aria-describedby'),a.getAttribute('title'))}var card=img;for(var d=0;d<8;d++){card=card.parentElement;if(!card)break;var allAttrs=[];for(var ai=0;ai<card.attributes.length;ai++){var at=card.attributes[ai];if(at.name!=='class'&&at.name!=='style'){allAttrs.push(at.name+'="'+at.value.substring(0,80)+'"')}}if(allAttrs.length>0){console.log('depth='+d+' <'+card.tagName.toLowerCase()+'> attrs: '+allAttrs.join(', '))}}var react=null;var el=img;for(var k in el){if(k.startsWith('__reactFiber')||k.startsWith('__reactInternalInstance')){react=el[k];break}}if(react){console.log('React fiber found!');var fiber=react;for(var rd=0;rd<10;rd++){if(!fiber)break;if(fiber.memoizedProps){var props=fiber.memoizedProps;var keys=Object.keys(props).filter(function(k){return k!=='children'&&k!=='style'&&k!=='className'});if(keys.length>0){console.log('  fiber depth='+rd+' props: '+keys.join(', '));for(var ki=0;ki<keys.length;ki++){var val=props[keys[ki]];if(typeof val==='string'&&val.length>0){console.log('    '+keys[ki]+'='+JSON.stringify(val).substring(0,150))}else if(typeof val==='object'&&val!==null){console.log('    '+keys[ki]+'='+JSON.stringify(val).substring(0,150))}}}}fiber=fiber.return}}console.log('---')}console.log('\n=== 완료 ===')})();
```
