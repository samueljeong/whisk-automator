# Flow DOM 심층 진단 #2 (수정)

```js
(function(){var imgs=document.querySelectorAll('img');var g=[];for(var i=0;i<imgs.length;i++){if(imgs[i].src&&imgs[i].src.includes('getMediaUrlRedirect')){g.push(imgs[i])}}console.log('=== 이미지 '+g.length+'개 심층 분석 ===');for(var gi=0;gi<Math.min(5,g.length);gi++){var img=g[gi];console.log('\n--- 이미지 #'+(gi+1)+' ---');console.log('FULL src:',img.src);console.log('alt:',img.alt||'(없음)');console.log('title:',img.title||'(없음)');var a=img.closest('a');if(a){console.log('링크 href:',a.href);console.log('링크 title:',a.getAttribute('title')||'(없음)');console.log('링크 aria-label:',a.getAttribute('aria-label')||'(없음)')}var react=null;var el=img;for(var k in el){if(k.startsWith('__reactFiber')||k.startsWith('__reactInternalInstance')){react=el[k];break}}if(react){console.log('React fiber found!');var fiber=react;for(var rd=0;rd<15;rd++){if(!fiber)break;if(fiber.memoizedProps){var props=fiber.memoizedProps;var keys=Object.keys(props).filter(function(k){return k!=='children'&&k!=='style'&&k!=='className'&&k!=='ref'});for(var ki=0;ki<keys.length;ki++){var val=props[keys[ki]];if(typeof val==='string'&&val.length>5){console.log('  fiber['+rd+'] '+keys[ki]+' = "'+val.substring(0,200)+'"')}else if(typeof val==='number'||typeof val==='boolean'){console.log('  fiber['+rd+'] '+keys[ki]+' = '+val)}}}fiber=fiber.return}}}console.log('\n=== 완료 ===')})();
```
