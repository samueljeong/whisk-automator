# trpcState.json 구조 탐색

```js
(function(){var imgs=document.querySelectorAll('img');var g=[];for(var i=0;i<imgs.length;i++){if(imgs[i].src&&imgs[i].src.includes('getMediaUrlRedirect')){g.push(imgs[i])}}var a=g[0].closest('a');var href=a.href;var editId=href.split('/edit/')[1];console.log('editId:',editId);fetch(href,{credentials:'include'}).then(function(r){return r.text()}).then(function(html){var m=html.match(/<script id="__NEXT_DATA__"[^>]*>([\s\S]*?)<\/script>/);var data=JSON.parse(m[1]);var json=data.props.pageProps.trpcState.json;console.log('json 키:',Object.keys(json));function explore(obj,path,depth){if(depth>4)return;if(!obj||typeof obj!=='object')return;var keys=Object.keys(obj);for(var i=0;i<keys.length;i++){var k=keys[i];var v=obj[k];var p=path+'.'+k;if(typeof v==='string'&&v.length>30&&!v.startsWith('http')&&!v.startsWith('/')&&!v.includes('<')){console.log(p+' = "'+v.substring(0,200)+'"')}else if(typeof v==='object'&&v!==null){if(Array.isArray(v)){console.log(p+' = Array['+v.length+']');if(v.length>0&&v.length<20){explore(v[0],p+'[0]',depth+1)}}else{var vk=Object.keys(v);console.log(p+' = {'+vk.slice(0,8).join(',')+'}');explore(v,p,depth+1)}}}}explore(json,'trpc',0);var str=JSON.stringify(json);console.log('\n전체 크기:',str.length);if(str.includes(editId)){var idx=str.indexOf(editId);console.log('editId 위치:',idx);console.log('주변 500자:',str.substring(Math.max(0,idx-200),idx+300))}}).catch(function(e){console.error('실패:',e)})})();
```
