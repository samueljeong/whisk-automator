# Flow API 프롬프트 추출 테스트

이미지의 edit URL에서 프롬프트를 가져올 수 있는지 확인합니다. Flow 탭 콘솔에 붙여넣기:

```js
(function(){var imgs=document.querySelectorAll('img');var g=[];for(var i=0;i<imgs.length;i++){if(imgs[i].src&&imgs[i].src.includes('getMediaUrlRedirect')){g.push(imgs[i])}}if(g.length===0){console.log('이미지 없음');return}var a=g[0].closest('a');if(!a){console.log('링크 없음');return}var href=a.href;var editId=href.split('/edit/')[1];console.log('테스트 이미지 edit URL:',href);console.log('edit ID:',editId);console.log('\n=== 방법1: edit 페이지 fetch ===');fetch(href,{credentials:'include'}).then(function(r){return r.text()}).then(function(html){console.log('HTML 길이:',html.length);var promptMatch=html.match(/prompt['":\s]*['"]([^'"]{20,})['"]/i);if(promptMatch){console.log('프롬프트 발견!',promptMatch[1].substring(0,200))}else{var scripts=html.match(/<script[^>]*>[\s\S]*?<\/script>/gi);console.log('script 태그 수:',scripts?scripts.length:0);var jsonMatch=html.match(/__NEXT_DATA__|__APP_DATA__|window\.__/g);console.log('데이터 패턴:',jsonMatch);var textSample=html.replace(/<[^>]*>/g,' ').replace(/\s+/g,' ').substring(0,500);console.log('텍스트 샘플:',textSample)}}).catch(function(e){console.error('fetch 실패:',e)});console.log('\n=== 방법2: tRPC API 시도 ===');var projectId=href.split('/project/')[1].split('/')[0];console.log('project ID:',projectId);var apiUrl='/fx/api/trpc/project.getProject?input='+encodeURIComponent(JSON.stringify({projectId:projectId}));console.log('API URL:',apiUrl);fetch(apiUrl,{credentials:'include'}).then(function(r){return r.json()}).then(function(data){console.log('API 응답 키:',Object.keys(data));var str=JSON.stringify(data);console.log('응답 길이:',str.length);if(str.includes(editId)){console.log('editId가 응답에 포함됨!');var idx=str.indexOf(editId);console.log('주변 텍스트:',str.substring(Math.max(0,idx-100),idx+200))}var prompts=str.match(/"prompt":"([^"]{20,})"/g);if(prompts){console.log('프롬프트 발견!',prompts.length+'개');for(var p=0;p<Math.min(5,prompts.length);p++){console.log('  ',prompts[p].substring(0,150))}}}).catch(function(e){console.error('API 실패:',e);console.log('다른 API 시도...');fetch('/fx/api/trpc/generation.getGeneration?input='+encodeURIComponent(JSON.stringify({generationId:editId})),{credentials:'include'}).then(function(r){return r.text()}).then(function(t){console.log('generation API 응답:',t.substring(0,500))}).catch(function(e2){console.error('generation API도 실패:',e2)})})})();
```
